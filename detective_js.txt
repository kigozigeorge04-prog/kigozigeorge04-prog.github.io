        let det_activeComplaintId = null;
        let det_scores = [];
        let det_selectedDiagnoses = [];
        let det_selectedOptions = [];
        let det_isFinished = false;
        let det_isOpen = false;
        let det_lastDiagnosis = '';
        let det_lastSummary = '';

        const DET_TAG_MAP = {
            'fever':'fever','cough':'cough','diarrhea':'diarrhoea','poor feeding':'poor_feeding',
            'convulsions':'seizure','rash':'rash','jaundice':'jaundice','pain':'abdominal_pain',
            'difficulty breathing':'difficulty_breathing','swelling':'oedema','lethargy':'lethargy',
            'pallor':'pallor','neck stiffness':'neck_stiffness','ear discharge':'ear_discharge',
            'oedema':'oedema','abdominal distension':'abdominal_pain','failure to thrive':'failure_to_thrive',
            'limping / bone pain':'limp','haematuria':'haematuria','dysuria / frequency':'dysuria',
            'delayed speech':'developmental_delay','eye problems':'eye_problems',
            'bleeding / bruising':'bleeding_bruising','unconsciousness':'unconsciousness',
            'malnutrition':'malnutrition','neonatal problems':'neonatal_problems'
        };

        function toggleDetectiveMode() {
            const panel = document.getElementById('detectivePanel');
            const btn = document.getElementById('detectiveModeBtn');
            det_isOpen = !det_isOpen;
            panel.classList.toggle('visible', det_isOpen);
            btn.classList.toggle('active', det_isOpen);
            btn.textContent = det_isOpen ? '✕ Close Detective' : '🕵️ Detective Mode';
            if (det_isOpen) {
                const selTag = document.querySelector('.tag-btn.selected');
                if (selTag) {
                    const key = DET_TAG_MAP[selTag.dataset.cat];
                    if (key && COMPLAINTS[key] && det_activeComplaintId !== key) detStartComplaint(key);
                }
                setTimeout(() => document.getElementById('detectivePanel').scrollIntoView({behavior:'smooth',block:'nearest'}), 150);
            }
        }

        // Wrap selectTag to trigger detective auto-start
        const _origSelectTag = window.selectTag;
        window.selectTag = function(btn) {
            _origSelectTag(btn);
            if (det_isOpen) {
                const key = DET_TAG_MAP[btn.dataset.cat];
                if (key && COMPLAINTS[key]) {
                    detStartComplaint(key);
                } else {
                    document.getElementById('detChatBox').innerHTML = '<div class="det-empty"><div class="det-big-icon">🤷</div><p>No detective case for this complaint yet.</p></div>';
                    document.getElementById('detOptionsContainer').innerHTML = '';
                    document.getElementById('detSummarySection').classList.remove('visible');
                }
            }
        };

        function detStartComplaint(id) {
            det_activeComplaintId = id;
            det_isFinished = false;
            const complaint = COMPLAINTS[id];
            det_selectedDiagnoses = complaint.diagnoses;
            det_scores = new Array(det_selectedDiagnoses.length).fill(0);
            det_selectedOptions = [];
            det_lastDiagnosis = ''; det_lastSummary = '';
            const chatBox = document.getElementById('detChatBox');
            chatBox.innerHTML = '';
            document.getElementById('detSummarySection').classList.remove('visible');
            document.getElementById('detOptionsContainer').innerHTML = '';
            detAddMessage('doctor', `🔍 Case: <strong>${complaint.label}</strong> — answer each question based on your patient.`);
            detShowQuestion(complaint, 0);
        }

        function detShowQuestion(complaint, index) {
            if (index >= complaint.questions.length) { detFinishDiagnosis(complaint); return; }
            const qData = complaint.questions[index];
            detAddMessage('doctor', qData.q, qData.hint);
            const optCon = document.getElementById('detOptionsContainer');
            optCon.innerHTML = '';
            qData.options.forEach((opt, i) => {
                const btn = document.createElement('button');
                btn.className = 'det-opt-btn';
                btn.textContent = opt.text;
                btn.addEventListener('click', () => detHandleOption(complaint, index, i));
                optCon.appendChild(btn);
            });
        }

        function detHandleOption(complaint, qIndex, optIndex) {
            const btns = document.querySelectorAll('.det-opt-btn');
            btns.forEach(b => b.disabled = true);
            btns[optIndex].classList.add('selected');
            const chosen = complaint.questions[qIndex].options[optIndex];
            det_selectedOptions.push({ text: chosen.text, scores: chosen.scores });
            detAddMessage('patient', chosen.text);
            chosen.scores.forEach((s, i) => { det_scores[i] = (det_scores[i] || 0) + (s || 0); });
            setTimeout(() => {
                if (qIndex + 1 < complaint.questions.length) detShowQuestion(complaint, qIndex + 1);
                else detFinishDiagnosis(complaint);
            }, 500);
        }

        function detFinishDiagnosis(complaint) {
            det_isFinished = true;
            document.getElementById('detOptionsContainer').innerHTML = '';
            let maxScore = -1, topIndex = 0;
            det_scores.forEach((s, i) => { if (s > maxScore) { maxScore = s; topIndex = i; } });
            const leadingDx = complaint.diagnoses[topIndex];
            det_lastDiagnosis = leadingDx;
            const positives = det_selectedOptions.filter(o => (o.scores[topIndex] || 0) > 0).map(o => o.text);
            const negatives = det_selectedOptions.filter(o => (o.scores[topIndex] || 0) === 0).map(o => o.text);
            const diffList = complaint.diagnoses.map((d, i) => {
                if (i === topIndex) return d + ' ✓ (leading)';
                return d + (det_scores[i] > 0 ? ` (${det_scores[i]} pts)` : ' (0 pts)');
            });
            const dxMsg = `**Potential Diagnosis: ${leadingDx}**\n\n**Positives:**\n${positives.length ? positives.map(p=>'- '+p).join('\n') : '- None'}\n\n**Negatives:**\n${negatives.length ? negatives.map(n=>'- '+n).join('\n') : '- None'}\n\n**Differentials:**\n${diffList.map(d=>'- '+d).join('\n')}`;
            detAddMessage('doctor', dxMsg, null, true);
            const childName = document.getElementById('childName')?.value?.trim() || 'child';
            const childAge = document.getElementById('childAge')?.value?.trim() || 'unknown age';
            const childSex = document.getElementById('childSex')?.value || '';
            let summary = `This is a ${childAge} ${childSex.toLowerCase()} (${childName}) presenting with ${complaint.label.toLowerCase()}.\n\nOn further questioning:\n`;
            summary += det_selectedOptions.map(o => '• ' + o.text).join('\n');
            summary += `\n\nPositive findings:\n${positives.map(p=>'• '+p).join('\n') || '• None'}`;
            summary += `\n\nNegative findings:\n${negatives.map(n=>'• '+n).join('\n') || '• None'}`;
            summary += `\n\nLeading diagnosis: ${leadingDx}\nDifferentials: ${diffList.join(' | ')}`;
            det_lastSummary = summary;
            document.getElementById('detSummaryBody').textContent = summary;
            document.getElementById('detSummarySection').classList.add('visible');
        }

        function detAddMessage(role, text, hint=null, isDiagnosis=false) {
            const chatBox = document.getElementById('detChatBox');
            const wrapper = document.createElement('div');
            wrapper.className = 'det-message ' + role;
            const bubble = document.createElement('div');
            bubble.className = 'det-bubble';
            const textSpan = document.createElement('span');
            textSpan.innerHTML = text.replace(/\n/g,'<br>');
            bubble.appendChild(textSpan);
            if (hint && role==='doctor' && !isDiagnosis) {
                const h = document.createElement('div');
                h.className = 'det-hint';
                h.textContent = hint;
                bubble.appendChild(h);
            }
            if (isDiagnosis) {
                const badge = document.createElement('div');
                badge.className = 'det-dx-badge';
                badge.innerHTML = '🧾 Diagnostic Impression';
                bubble.appendChild(badge);
                const sdx = document.createElement('div');
                sdx.className = 'det-structured-dx';
                sdx.innerHTML = text.replace(/\n/g,'<br>');
                bubble.appendChild(sdx);
            }
            wrapper.appendChild(bubble);
            const meta = document.createElement('div');
            meta.className = 'det-meta';
            meta.textContent = role==='doctor' ? '🩺 Dr.' : '🧑 You';
            wrapper.appendChild(meta);
            chatBox.appendChild(wrapper);
            setTimeout(() => { chatBox.scrollTop = chatBox.scrollHeight; }, 50);
        }

        function detCopySummary() {
            navigator.clipboard.writeText(det_lastSummary || '').then(() => showToast('📋 Summary copied!','#0a2942'));
        }

        function applyDetectiveDiagnosis() {
            if (!det_lastDiagnosis) { showToast('Complete the detective session first.','#e67e22'); return; }
            const hpc = document.getElementById('hpcNarrative');
            if (hpc) hpc.value = (hpc.value ? det_lastSummary + '\n\n---\n\n' + hpc.value : det_lastSummary);
            const assoc = document.getElementById('associatedComplaints');
            if (assoc) assoc.value = (assoc.value ? assoc.value + '\n' : '') + '[Detective] Leading Dx: ' + det_lastDiagnosis;
            showToast('✅ Applied to HPC narrative!','#1a7a4a');
            updateProgress();
        }
